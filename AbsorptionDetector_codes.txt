// This Pine Script™ code is subject to the terms of the Mozilla Public License 2.0 at https://mozilla.org/MPL/2.0/

//@version=5
indicator("Absorption Detector", shorttitle="AbsDetect", overlay=true)

// =============================================================================
// INPUTS
// =============================================================================

// Volume Settings
volumeSpikeMultiplier = input.float(3.0, title="Volume Spike Multiplier (Weak)", minval=1.0, step=0.1,
    tooltip="Minimum volume multiplier for weak absorption signal", group="Volume Settings")
volumeSpikeMedium = input.float(5.0, title="Volume Spike Multiplier (Medium)", minval=1.0, step=0.1,
    tooltip="Volume multiplier threshold for medium absorption signal", group="Volume Settings")
volumeSpikeStrong = input.float(8.0, title="Volume Spike Multiplier (Strong)", minval=1.0, step=0.1,
    tooltip="Volume multiplier threshold for strong absorption signal", group="Volume Settings")
volumeSmaLength = input.int(20, title="Volume SMA Length", minval=1,
    tooltip="Period for calculating average volume", group="Volume Settings")

// Price Settings
priceChangeThreshold = input.float(0.2, title="Price Change Threshold (%)", minval=0.01, step=0.01,
    tooltip="Maximum body size as percentage of open price to be considered flat", group="Price Settings")

// Pressure Ratio Settings
pressureRatio = input.float(1.5, title="Pressure Ratio", minval=1.0, step=0.1,
    tooltip="Buy/Sell volume ratio threshold for absorption detection", group="Pressure Settings")

// Display Settings
showSignals = input.bool(true, title="Show Signals",
    tooltip="Toggle visibility of absorption signals on chart", group="Display")
showBackground = input.bool(true, title="Show Background Highlight",
    tooltip="Toggle background color on absorption bars", group="Display")

// Alert Settings
alertEnabled = input.bool(false, title="Alert Enabled",
    tooltip="Enable alert notifications for absorption signals", group="Alerts")

// =============================================================================
// COLOR SETTINGS
// =============================================================================

// Buy Absorption Colors (Green gradient)
colorBuyWeak = input.color(color.new(#4CAF50, 60), title="Buy Weak", group="Colors - Buy Absorption")
colorBuyMedium = input.color(color.new(#2E7D32, 30), title="Buy Medium", group="Colors - Buy Absorption")
colorBuyStrong = input.color(color.new(#1B5E20, 0), title="Buy Strong", group="Colors - Buy Absorption")

// Sell Absorption Colors (Red gradient)
colorSellWeak = input.color(color.new(#F44336, 60), title="Sell Weak", group="Colors - Sell Absorption")
colorSellMedium = input.color(color.new(#C62828, 30), title="Sell Medium", group="Colors - Sell Absorption")
colorSellStrong = input.color(color.new(#B71C1C, 0), title="Sell Strong", group="Colors - Sell Absorption")

// =============================================================================
// CALCULATIONS
// =============================================================================

// Volume Analysis
volAvg = ta.sma(volume, volumeSmaLength)
volRatio = volume / volAvg

// Volume Spike Levels
volSpikeWeak = volRatio >= volumeSpikeMultiplier
volSpikeMedium = volRatio >= volumeSpikeMedium
volSpikeStrong = volRatio >= volumeSpikeStrong

// Price Flat Detection (horizontal price movement)
priceChangePct = math.abs(close - open) / open * 100
priceFlat = priceChangePct < priceChangeThreshold

// Buy/Sell Volume Estimation
// Using price position within bar to estimate buy/sell pressure
barRange = high - low + 1e-6  // Prevent division by zero
buyVol = volume * ((close - low) / barRange)
sellVol = volume - buyVol

// Dominance Detection
buyDominant = buyVol > sellVol * pressureRatio
sellDominant = sellVol > buyVol * pressureRatio

// Valid bar check (skip zero volume bars)
validBar = volume > 0

// =============================================================================
// ABSORPTION SIGNALS
// =============================================================================

// Buy Absorption (bullish accumulation)
buyAbsorptionWeak = volSpikeWeak and priceFlat and buyDominant and validBar
buyAbsorptionMedium = volSpikeMedium and priceFlat and buyDominant and validBar
buyAbsorptionStrong = volSpikeStrong and priceFlat and buyDominant and validBar

// Sell Absorption (bearish distribution)
sellAbsorptionWeak = volSpikeWeak and priceFlat and sellDominant and validBar
sellAbsorptionMedium = volSpikeMedium and priceFlat and sellDominant and validBar
sellAbsorptionStrong = volSpikeStrong and priceFlat and sellDominant and validBar

// Combined signals (any level)
buyAbsorption = buyAbsorptionWeak
sellAbsorption = sellAbsorptionWeak

// =============================================================================
// SIGNAL STRENGTH DETERMINATION
// =============================================================================

// Determine signal strength level (3=strong, 2=medium, 1=weak, 0=none)
buyStrength = buyAbsorptionStrong ? 3 : buyAbsorptionMedium ? 2 : buyAbsorptionWeak ? 1 : 0
sellStrength = sellAbsorptionStrong ? 3 : sellAbsorptionMedium ? 2 : sellAbsorptionWeak ? 1 : 0

// Get appropriate color based on strength
getBuyColor(strength) =>
    strength == 3 ? colorBuyStrong : strength == 2 ? colorBuyMedium : colorBuyWeak

getSellColor(strength) =>
    strength == 3 ? colorSellStrong : strength == 2 ? colorSellMedium : colorSellWeak

buySignalColor = getBuyColor(buyStrength)
sellSignalColor = getSellColor(sellStrength)

// =============================================================================
// OUTPUTS - SIGNAL VISUALIZATION
// =============================================================================

// Buy Absorption Signals (triangles below bar)
plotshape(showSignals and buyAbsorptionStrong,
    title="Buy Absorption Strong",
    style=shape.triangleup,
    location=location.belowbar,
    color=colorBuyStrong,
    size=size.normal,
    text="Abs")

plotshape(showSignals and buyAbsorptionMedium and not buyAbsorptionStrong,
    title="Buy Absorption Medium",
    style=shape.triangleup,
    location=location.belowbar,
    color=colorBuyMedium,
    size=size.small,
    text="Abs")

plotshape(showSignals and buyAbsorptionWeak and not buyAbsorptionMedium,
    title="Buy Absorption Weak",
    style=shape.triangleup,
    location=location.belowbar,
    color=colorBuyWeak,
    size=size.tiny,
    text="Abs")

// Sell Absorption Signals (triangles above bar)
plotshape(showSignals and sellAbsorptionStrong,
    title="Sell Absorption Strong",
    style=shape.triangledown,
    location=location.abovebar,
    color=colorSellStrong,
    size=size.normal,
    text="Abs")

plotshape(showSignals and sellAbsorptionMedium and not sellAbsorptionStrong,
    title="Sell Absorption Medium",
    style=shape.triangledown,
    location=location.abovebar,
    color=colorSellMedium,
    size=size.small,
    text="Abs")

plotshape(showSignals and sellAbsorptionWeak and not sellAbsorptionMedium,
    title="Sell Absorption Weak",
    style=shape.triangledown,
    location=location.abovebar,
    color=colorSellWeak,
    size=size.tiny,
    text="Abs")

// =============================================================================
// OUTPUTS - BACKGROUND HIGHLIGHT
// =============================================================================

// Background colors with strength-based opacity
bgBuyColor = buyAbsorptionStrong ? color.new(colorBuyStrong, 85) :
             buyAbsorptionMedium ? color.new(colorBuyMedium, 90) :
             buyAbsorptionWeak ? color.new(colorBuyWeak, 93) : na

bgSellColor = sellAbsorptionStrong ? color.new(colorSellStrong, 85) :
              sellAbsorptionMedium ? color.new(colorSellMedium, 90) :
              sellAbsorptionWeak ? color.new(colorSellWeak, 93) : na

bgcolor(showBackground and buyAbsorption ? bgBuyColor : na, title="Buy Absorption Background")
bgcolor(showBackground and sellAbsorption ? bgSellColor : na, title="Sell Absorption Background")

// =============================================================================
// OUTPUTS - ALERTS (Bar Close Only)
// =============================================================================

// Buy Absorption Alerts
alertcondition(buyAbsorptionWeak,
    title="Buy Absorption (Any)",
    message="Buy Absorption Detected - Bullish accumulation signal")

alertcondition(buyAbsorptionMedium,
    title="Buy Absorption (Medium+)",
    message="Medium Buy Absorption Detected - Significant bullish accumulation")

alertcondition(buyAbsorptionStrong,
    title="Buy Absorption (Strong)",
    message="Strong Buy Absorption Detected - Major bullish accumulation!")

// Sell Absorption Alerts
alertcondition(sellAbsorptionWeak,
    title="Sell Absorption (Any)",
    message="Sell Absorption Detected - Bearish distribution signal")

alertcondition(sellAbsorptionMedium,
    title="Sell Absorption (Medium+)",
    message="Medium Sell Absorption Detected - Significant bearish distribution")

alertcondition(sellAbsorptionStrong,
    title="Sell Absorption (Strong)",
    message="Strong Sell Absorption Detected - Major bearish distribution!")

// Combined Alert
alertcondition(buyAbsorption or sellAbsorption,
    title="Any Absorption",
    message="Absorption Signal Detected")

// =============================================================================
// INFO TABLE (Optional - Shows current status)
// =============================================================================

var table infoTable = table.new(position.top_right, 2, 4, bgcolor=color.new(color.black, 80))

if barstate.islast
    table.cell(infoTable, 0, 0, "Vol Ratio", text_color=color.white, text_size=size.small)
    table.cell(infoTable, 1, 0, str.tostring(volRatio, "#.##") + "x", text_color=color.yellow, text_size=size.small)
    table.cell(infoTable, 0, 1, "Buy Vol %", text_color=color.white, text_size=size.small)
    table.cell(infoTable, 1, 1, str.tostring(buyVol / volume * 100, "#.#") + "%", text_color=color.green, text_size=size.small)
    table.cell(infoTable, 0, 2, "Sell Vol %", text_color=color.white, text_size=size.small)
    table.cell(infoTable, 1, 2, str.tostring(sellVol / volume * 100, "#.#") + "%", text_color=color.red, text_size=size.small)
    table.cell(infoTable, 0, 3, "Price Chg", text_color=color.white, text_size=size.small)
    table.cell(infoTable, 1, 3, str.tostring(priceChangePct, "#.##") + "%", text_color=color.white, text_size=size.small)

// This Pine Script™ code is subject to the terms of the Mozilla Public License 2.0 at https://mozilla.org/MPL/2.0/

//@version=5
strategy("Absorption Detector Strategy", shorttitle="AbsStrategy", overlay=true,
    default_qty_type=strategy.percent_of_equity,
    default_qty_value=100,
    initial_capital=10000,
    commission_type=strategy.commission.percent,
    commission_value=0.1,
    slippage=1)

// =============================================================================
// INPUTS
// =============================================================================

// Volume Settings
volumeSpikeMultiplier = input.float(3.0, title="Volume Spike Multiplier (Weak)", minval=1.0, step=0.1,
    tooltip="Minimum volume multiplier for weak absorption signal", group="Volume Settings")
volumeSpikeMedium = input.float(5.0, title="Volume Spike Multiplier (Medium)", minval=1.0, step=0.1,
    tooltip="Volume multiplier threshold for medium absorption signal", group="Volume Settings")
volumeSpikeStrong = input.float(8.0, title="Volume Spike Multiplier (Strong)", minval=1.0, step=0.1,
    tooltip="Volume multiplier threshold for strong absorption signal", group="Volume Settings")
volumeSmaLength = input.int(20, title="Volume SMA Length", minval=1,
    tooltip="Period for calculating average volume", group="Volume Settings")

// Price Settings
priceChangeThreshold = input.float(0.2, title="Price Change Threshold (%)", minval=0.01, step=0.01,
    tooltip="Maximum body size as percentage of open price to be considered flat", group="Price Settings")

// Pressure Ratio Settings
pressureRatio = input.float(1.5, title="Pressure Ratio", minval=1.0, step=0.1,
    tooltip="Buy/Sell volume ratio threshold for absorption detection", group="Pressure Settings")

// Strategy Settings
minSignalStrength = input.string("Weak", title="Minimum Signal Strength",
    options=["Weak", "Medium", "Strong"],
    tooltip="Minimum absorption strength required for entry", group="Strategy Settings")
useStopLoss = input.bool(true, title="Use Stop Loss", group="Strategy Settings")
stopLossPercent = input.float(2.0, title="Stop Loss (%)", minval=0.1, step=0.1, group="Strategy Settings")
useTakeProfit = input.bool(true, title="Use Take Profit", group="Strategy Settings")
takeProfitPercent = input.float(4.0, title="Take Profit (%)", minval=0.1, step=0.1, group="Strategy Settings")
useTrailingStop = input.bool(false, title="Use Trailing Stop", group="Strategy Settings")
trailingStopPercent = input.float(1.5, title="Trailing Stop (%)", minval=0.1, step=0.1, group="Strategy Settings")
maxBarsInTrade = input.int(20, title="Max Bars in Trade (0=disabled)", minval=0, group="Strategy Settings")
exitOnOppositeSignal = input.bool(true, title="Exit on Opposite Signal", group="Strategy Settings")

// Trade Direction
tradeDirection = input.string("Both", title="Trade Direction",
    options=["Long Only", "Short Only", "Both"], group="Strategy Settings")

// Display Settings
showSignals = input.bool(true, title="Show Signals", group="Display")
showBackground = input.bool(true, title="Show Background Highlight", group="Display")

// =============================================================================
// COLOR SETTINGS
// =============================================================================

colorBuyWeak = input.color(color.new(#4CAF50, 60), title="Buy Weak", group="Colors - Buy")
colorBuyMedium = input.color(color.new(#2E7D32, 30), title="Buy Medium", group="Colors - Buy")
colorBuyStrong = input.color(color.new(#1B5E20, 0), title="Buy Strong", group="Colors - Buy")

colorSellWeak = input.color(color.new(#F44336, 60), title="Sell Weak", group="Colors - Sell")
colorSellMedium = input.color(color.new(#C62828, 30), title="Sell Medium", group="Colors - Sell")
colorSellStrong = input.color(color.new(#B71C1C, 0), title="Sell Strong", group="Colors - Sell")

// =============================================================================
// CALCULATIONS
// =============================================================================

// Volume Analysis
volAvg = ta.sma(volume, volumeSmaLength)
volRatio = volume / volAvg

// Volume Spike Levels
volSpikeWeak = volRatio >= volumeSpikeMultiplier
volSpikeMedium = volRatio >= volumeSpikeMedium
volSpikeStrong = volRatio >= volumeSpikeStrong

// Price Flat Detection
priceChangePct = math.abs(close - open) / open * 100
priceFlat = priceChangePct < priceChangeThreshold

// Buy/Sell Volume Estimation
barRange = high - low + 1e-6
buyVol = volume * ((close - low) / barRange)
sellVol = volume - buyVol

// Dominance Detection
buyDominant = buyVol > sellVol * pressureRatio
sellDominant = sellVol > buyVol * pressureRatio

// Valid bar check
validBar = volume > 0

// =============================================================================
// ABSORPTION SIGNALS
// =============================================================================

// Buy Absorption levels
buyAbsorptionWeak = volSpikeWeak and priceFlat and buyDominant and validBar
buyAbsorptionMedium = volSpikeMedium and priceFlat and buyDominant and validBar
buyAbsorptionStrong = volSpikeStrong and priceFlat and buyDominant and validBar

// Sell Absorption levels
sellAbsorptionWeak = volSpikeWeak and priceFlat and sellDominant and validBar
sellAbsorptionMedium = volSpikeMedium and priceFlat and sellDominant and validBar
sellAbsorptionStrong = volSpikeStrong and priceFlat and sellDominant and validBar

// Signal strength determination
buyStrength = buyAbsorptionStrong ? 3 : buyAbsorptionMedium ? 2 : buyAbsorptionWeak ? 1 : 0
sellStrength = sellAbsorptionStrong ? 3 : sellAbsorptionMedium ? 2 : sellAbsorptionWeak ? 1 : 0

// Minimum strength filter
minStrengthValue = minSignalStrength == "Strong" ? 3 : minSignalStrength == "Medium" ? 2 : 1

// Entry signals based on minimum strength
buySignal = buyStrength >= minStrengthValue
sellSignal = sellStrength >= minStrengthValue

// =============================================================================
// STRATEGY LOGIC
// =============================================================================

// Track bars since entry
var int barsSinceEntry = 0
if strategy.position_size != 0
    barsSinceEntry := barsSinceEntry + 1
else
    barsSinceEntry := 0

// Entry conditions
canGoLong = tradeDirection == "Long Only" or tradeDirection == "Both"
canGoShort = tradeDirection == "Short Only" or tradeDirection == "Both"

longCondition = buySignal and canGoLong and strategy.position_size <= 0
shortCondition = sellSignal and canGoShort and strategy.position_size >= 0

// Exit conditions
timeExit = maxBarsInTrade > 0 and barsSinceEntry >= maxBarsInTrade
oppositeExitLong = exitOnOppositeSignal and sellSignal and strategy.position_size > 0
oppositeExitShort = exitOnOppositeSignal and buySignal and strategy.position_size < 0

// =============================================================================
// EXECUTE TRADES
// =============================================================================

// Long Entry
if longCondition
    strategy.entry("Long", strategy.long)

// Short Entry
if shortCondition
    strategy.entry("Short", strategy.short)

// Stop Loss and Take Profit
if strategy.position_size > 0  // Long position
    stopPrice = strategy.position_avg_price * (1 - stopLossPercent / 100)
    takePrice = strategy.position_avg_price * (1 + takeProfitPercent / 100)

    if useStopLoss and useTakeProfit
        strategy.exit("Long Exit", "Long", stop=stopPrice, limit=takePrice,
            trail_points=useTrailingStop ? close * trailingStopPercent / 100 / syminfo.mintick : na,
            trail_offset=useTrailingStop ? close * trailingStopPercent / 100 / syminfo.mintick : na)
    else if useStopLoss
        strategy.exit("Long SL", "Long", stop=stopPrice)
    else if useTakeProfit
        strategy.exit("Long TP", "Long", limit=takePrice)

if strategy.position_size < 0  // Short position
    stopPrice = strategy.position_avg_price * (1 + stopLossPercent / 100)
    takePrice = strategy.position_avg_price * (1 - takeProfitPercent / 100)

    if useStopLoss and useTakeProfit
        strategy.exit("Short Exit", "Short", stop=stopPrice, limit=takePrice,
            trail_points=useTrailingStop ? close * trailingStopPercent / 100 / syminfo.mintick : na,
            trail_offset=useTrailingStop ? close * trailingStopPercent / 100 / syminfo.mintick : na)
    else if useStopLoss
        strategy.exit("Short SL", "Short", stop=stopPrice)
    else if useTakeProfit
        strategy.exit("Short TP", "Short", limit=takePrice)

// Time-based exit
if timeExit
    strategy.close_all("Time Exit")

// Opposite signal exit
if oppositeExitLong
    strategy.close("Long", comment="Opposite Signal")
if oppositeExitShort
    strategy.close("Short", comment="Opposite Signal")

// =============================================================================
// VISUALIZATION
// =============================================================================

// Buy Absorption Signals
plotshape(showSignals and buyAbsorptionStrong,
    title="Buy Strong", style=shape.triangleup, location=location.belowbar,
    color=colorBuyStrong, size=size.normal, text="Abs")

plotshape(showSignals and buyAbsorptionMedium and not buyAbsorptionStrong,
    title="Buy Medium", style=shape.triangleup, location=location.belowbar,
    color=colorBuyMedium, size=size.small, text="Abs")

plotshape(showSignals and buyAbsorptionWeak and not buyAbsorptionMedium,
    title="Buy Weak", style=shape.triangleup, location=location.belowbar,
    color=colorBuyWeak, size=size.tiny, text="Abs")

// Sell Absorption Signals
plotshape(showSignals and sellAbsorptionStrong,
    title="Sell Strong", style=shape.triangledown, location=location.abovebar,
    color=colorSellStrong, size=size.normal, text="Abs")

plotshape(showSignals and sellAbsorptionMedium and not sellAbsorptionStrong,
    title="Sell Medium", style=shape.triangledown, location=location.abovebar,
    color=colorSellMedium, size=size.small, text="Abs")

plotshape(showSignals and sellAbsorptionWeak and not sellAbsorptionMedium,
    title="Sell Weak", style=shape.triangledown, location=location.abovebar,
    color=colorSellWeak, size=size.tiny, text="Abs")

// Background
bgBuyColor = buyAbsorptionStrong ? color.new(colorBuyStrong, 85) :
             buyAbsorptionMedium ? color.new(colorBuyMedium, 90) :
             buyAbsorptionWeak ? color.new(colorBuyWeak, 93) : na

bgSellColor = sellAbsorptionStrong ? color.new(colorSellStrong, 85) :
              sellAbsorptionMedium ? color.new(colorSellMedium, 90) :
              sellAbsorptionWeak ? color.new(colorSellWeak, 93) : na

bgcolor(showBackground and buyAbsorptionWeak ? bgBuyColor : na, title="Buy Background")
bgcolor(showBackground and sellAbsorptionWeak ? bgSellColor : na, title="Sell Background")

// =============================================================================
// PERFORMANCE TABLE
// =============================================================================

var table perfTable = table.new(position.top_right, 2, 6, bgcolor=color.new(color.black, 80))

if barstate.islast
    winRate = strategy.wintrades / math.max(strategy.closedtrades, 1) * 100
    profitFactor = strategy.grossprofit / math.max(strategy.grossloss, 1)

    table.cell(perfTable, 0, 0, "Total Trades", text_color=color.white, text_size=size.small)
    table.cell(perfTable, 1, 0, str.tostring(strategy.closedtrades), text_color=color.yellow, text_size=size.small)

    table.cell(perfTable, 0, 1, "Win Rate", text_color=color.white, text_size=size.small)
    table.cell(perfTable, 1, 1, str.tostring(winRate, "#.#") + "%",
        text_color=winRate >= 50 ? color.green : color.red, text_size=size.small)

    table.cell(perfTable, 0, 2, "Profit Factor", text_color=color.white, text_size=size.small)
    table.cell(perfTable, 1, 2, str.tostring(profitFactor, "#.##"),
        text_color=profitFactor >= 1 ? color.green : color.red, text_size=size.small)

    table.cell(perfTable, 0, 3, "Net Profit", text_color=color.white, text_size=size.small)
    table.cell(perfTable, 1, 3, str.tostring(strategy.netprofit, "#.##"),
        text_color=strategy.netprofit >= 0 ? color.green : color.red, text_size=size.small)

    table.cell(perfTable, 0, 4, "Max DD", text_color=color.white, text_size=size.small)
    table.cell(perfTable, 1, 4, str.tostring(strategy.max_drawdown, "#.##"), text_color=color.red, text_size=size.small)

    table.cell(perfTable, 0, 5, "Vol Ratio", text_color=color.white, text_size=size.small)
    table.cell(perfTable, 1, 5, str.tostring(volRatio, "#.##") + "x", text_color=color.yellow, text_size=size.small)
